        t"""
        ul(class="list", css="color: rebeccapurple; margin: 1px;"):
            if {flag}:
                li(id="first"): "one"
            elif {mode}:
                li: "two"
            else:
                li: "three"
            for item in {items}:

                Card(title="row"):
                    {label}
    """,
    )

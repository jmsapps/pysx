from pysx import Binding, defer, pysx

def view():
    row = Binding[int]("row")
    return pysx(rt"""
        if {visible}:
          p: "first"
        elif {other}:
          p: 'second'
        else:
          p: "last"
        match {mode}:
          case "a" | "b":
            span: "pair"
          case _:
            span: "other"
        for row in {rows} key={row}:
          let label = {defer(row, str)}
          set label = {defer(row, lambda n: f"row-{n}")}
          discard {defer(row, print)}
          Card(
            title={label}
            data-row={row},
          ): "literal {{braces}}"
          {tuple(pysx(t'\nspan: {n}') for n in range(2))}
    """, use=(Card,), namespace={"row": row})

single = pysx(t'''
    section: 'single quoted'
''')
